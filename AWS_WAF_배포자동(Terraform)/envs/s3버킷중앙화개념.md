구성도
설명: S3 버킷은 RAM(Resource Access Manager)으로 구성할 수 없음. 아래와 같은 구조로 Cross-Account 접근을 위한 버킷 정책으로 구현필요
<img width="1204" height="33" alt="image" src="https://github.com/user-attachments/assets/2ee48c0c-6cf7-47a0-9777-e1e36d9c1df2" />
   
사전 요구 조건
		a. Log Archive 계정에서 WAF 로그 중앙화용 S3 버킷 생성. 이때 버킷 이름은 다음 규칙을 따라야함 ex)aws-waf-logs-*
		b. 생성한 S3 버킷에서 아래와 같이 버킷 권한 정책 추가
		{
		    "Version": "2012-10-17",
		    "Statement": [
		        {
		            "Sid": "WAFDirectPut",
		            "Effect": "Allow",
		            "Principal": {
		                "Service": "wafv2.amazonaws.com"
		            },
		            "Action": "s3:PutObject",
		            "Resource": "arn:aws:s3:::aws-waf-logs-jaehwan251124/AWSLogs/*", #생성한 S3 버킷 ARN입력
		            "Condition": {
		                "StringEquals": {
		                    "aws:SourceAccount": [
		                        "962889423329", #WAF 연동한 계정 AccountID 입력(이후 계속 추가)
		                        "419893468851" #WAF 연동한 계정 AccountID 입력(이후 계속 추가)
		                    ],
		                    "s3:x-amz-acl": "bucket-owner-full-control"
		                }
		            }
		        },
		        {
		            "Sid": "WAFGetBucketAcl",
		            "Effect": "Allow",
		            "Principal": {
		                "Service": "wafv2.amazonaws.com"
		            },
		            "Action": "s3:GetBucketAcl",
		            "Resource": "arn:aws:s3:::aws-waf-logs-jaehwan251124", #생성한 S3 버킷 ARN입력
		            "Condition": {
		                "StringEquals": {
		                    "aws:SourceAccount": [
		                        "962889423329", #WAF 연동한 계정 AccountID 입력(이후 계속 추가)
		                        "419893468851" #WAF 연동한 계정 AccountID 입력(이후 계속 추가)
		                    ]
		                }
		            }
		        },
		        {
		            "Sid": "AWSLogDeliveryWrite",
		            "Effect": "Allow",
		            "Principal": {
		                "Service": "delivery.logs.amazonaws.com"
		            },
		            "Action": "s3:PutObject",
		            "Resource": "arn:aws:s3:::aws-waf-logs-jaehwan251124/AWSLogs/*", #생성한 S3 버킷 ARN입력
		            "Condition": {
		                "StringEquals": {
		                    "aws:SourceAccount": [
		                        "962889423329", #WAF 연동한 계정 AccountID 입력(이후 계속 추가)
		                        "419893468851" #WAF 연동한 계정 AccountID 입력(이후 계속 추가)
		                    ],
		                    "s3:x-amz-acl": "bucket-owner-full-control"
		                }
		            }
		        },
		        {
		            "Sid": "AWSLogDeliveryAclCheck",
		            "Effect": "Allow",
		            "Principal": {
		                "Service": "delivery.logs.amazonaws.com"
		            },
		            "Action": "s3:GetBucketAcl",
		            "Resource": "arn:aws:s3:::aws-waf-logs-jaehwan251124", #생성한 S3 버킷 ARN입력
		            "Condition": {
		                "StringEquals": {
		                    "aws:SourceAccount": [
		                        "962889423329", #WAF 연동한 계정 AccountID 입력(이후 계속 추가)
		                        "419893468851" #WAF 연동한 계정 AccountID 입력(이후 계속 추가)
		                    ]
		                }
		            }
		        }
		    ]
		}
